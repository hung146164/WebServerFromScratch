-- ============================================================
--  TRIGGER: CẬP NHẬT THỜI GIAN
-- ============================================================

CREATE TRIGGER trg_nguoidung_update_time ON NGUOI_DUNG AFTER UPDATE AS
BEGIN
    SET NOCOUNT ON;
    UPDATE NGUOI_DUNG SET cap_nhat_luc = GETDATE() FROM NGUOI_DUNG nd JOIN inserted i ON nd.nguoi_dung_id = i.nguoi_dung_id;
END;
GO

CREATE TRIGGER trg_hocphiky_update_time ON HOC_PHI_KY AFTER UPDATE AS
BEGIN
    SET NOCOUNT ON;
    UPDATE HOC_PHI_KY SET cap_nhat_luc = GETDATE() FROM HOC_PHI_KY hpk JOIN inserted i ON hpk.hoc_phi_ky_id = i.hoc_phi_ky_id;
END;
GO

CREATE TRIGGER trg_quangcao_update_time ON QUANG_CAO AFTER UPDATE AS
BEGIN
    SET NOCOUNT ON;
    UPDATE QUANG_CAO SET cap_nhat_luc = GETDATE() FROM QUANG_CAO qc JOIN inserted i ON qc.quang_cao_id = i.quang_cao_id;
END;    
GO
